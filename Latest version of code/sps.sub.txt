DEF  SPS ( )


  ;FOLD DECLARATIONS
  ;FOLD BASISTECH DECL
  ;Automatik extern
  DECL STATE_T STAT
  DECL MODUS_T MODE


  decl int count1
  DECL INT TimeSlow
  DECL INT TimeFast  
  DECL INT TimeNormal
  DECL INT TimeUltraSlow
  DECL INT TIMEIntermidiate
  DECL BOOL Aanvraag
  Decl INT AanvraagCount 
  DECL INT outputs

  
  outputs=0
  count1 = 0
  
  Aanvraag = false
   
  AanvraagCount =0
  
  TimeUltraSlow = 5000
  TimeSlow =3000
  TIMEIntermidiate =1500
  TimeNormal = 1000
  TimeFast = 100
  ;ENDFOLD (BASISTECH DECL)
  ;FOLD USER DECL
  ; Please insert user defined declarations

  ;ENDFOLD (USER DECL)
  ;ENDFOLD (DECLARATIONS)
  ;FOLD INI
  ;FOLD BASISTECH INIT
  BasisTech_PLC_INIT()
  ;ENDFOLD (BASISTECH INIT)
  ;FOLD AUTOEXT INIT
  INTERRUPT DECL 91 WHEN $PRO_STATE1==#P_FREE DO RESET_OUT ()
  INTERRUPT ON 91
  INTERRUPT DECL 92 WHEN $PRO_MOVE==TRUE DO RESET_LINESEL()
  INTERRUPT ON 92
  $LOOP_MSG[]="                                                            "
  MODE=#SYNC
  $H_POS=$H_POS
  ;Automatik extern
  IF $MODE_OP==#EX THEN
    CWRITE($CMD,STAT,MODE,"RUN /R1/CELL()")
  ENDIF
  ;ENDFOLD (AUTOEXT INIT)
  ;FOLD BACKUPMANAGER PLC INIT
  BM_ENABLED = FALSE
  BM_OUTPUTVALUE = 0
  ;ENDFOLD (BACKUPMANAGER PLC INIT)
  ;FOLD USER INIT
  ; Please insert user defined initialization commands

  ;ENDFOLD (USER INIT)
  ;ENDFOLD (INI)

  LOOP
    WAIT FOR NOT($POWER_FAIL)
	;FOLD BASISTECH PLC
	   BasisTech_PLC_LOOP()
	   
	;snelheidssturing
 IF ($IN[14]== TRUE) AND (BTN_OLD == FALSE) THEN ;als de witte knop wordt ingedrukt (met flank controle)
Speed_Level = Speed_Level + 1 ;wordt de volgende snelheid gekozen

IF Speed_Level > 3 THEN ;na stand 3 gaat hij terug naar stand 1
Speed_Level = 1
ENDIF
ENDIF

BTN_OLD = $IN[14] ;schrijf toestand drukknop naar variabele ter vergelijking voor flankdetectie

SWITCH Speed_Level
   CASE 1
      $OV_PRO = 30 ;override snelheid naar 30%
   CASE 2
      $OV_PRO = 70 ;override snelheid naar 70%
   CASE 3
      $OV_PRO = 100 ;override snelheid naar 100%
ENDSWITCH


; dit is het deel om de robot te laten starten en stoppen
if $in[5] then
   pulse ($OUT[20],true,1)
    bWerk = true
      $OUT[4] = TRUE;
  $OUT [6] = FALSE;
endif

if not $in[4] then
   bWerk =false
   $OUT[6] = TRUE;
   $OUT[4] = FALSE;
endif
  
IF NOT BWerk then
   $OUT[6] = TRUE;
   $OUT[4] = FALSE;
else 
   $OUT[4] = TRUE;
  $OUT [6] = FALSE;
ENDIF
  
  
; modus beveiligignen en signalen

IF $T1 then
   $OUT[5] =true           ;Move enable 
   $OUT[1] = FALSE         ;Doorlock los
   

   $OUT[11]=false

ENDIF

IF $T2 THEN
   $OUT[1] = FALSE         ;DOORLOCK open
   $OUT[5] =false          ;Move enable 

ENDIF
   
IF  $AUT THEN
    $OUT[5] =true          ;Move enable 
   
   IF NOT $IN[1] THEN      ; SAFFETYRELAY DOOR OKE
      $OUT[2]=TRUE         ; RESET LAMP AAN
      IF $IN[2] THEN       ; RESET BUTTON PUSHED?
         $OUT[1] =TRUE     ; DOORLOCK active
         PULSE( $OUT[14], TRUE,2)      ; reset floorscanner (not yet in use)
         PULSE( $OUT[15],TRUE,2)       ; reset safety matt
         $OUT[3]=false
         pulse( $OUT[19], TRUE,0.10)
         $OUT[20] = true


      ENDIF
      ENDIF
      
    if $IN[3] THEN
      Aanvraag = true
      bWerk = false
      
      
    ENDIF
    
    IF Aanvraag and NOT runningAprogram then
      $OUT[1]=false
      $OUT[3] =False
      Aanvraag =false
    endiF

ENDIF

IF  $EXT THEN
      
    $OUT[5] =true          ;Move enable 
   
   IF NOT $IN[1] THEN      ; SAFFETYRELAY DOOR OKE
      $OUT[2]=TRUE         ; RESET LAMP AAN
      IF $IN[2] THEN       ; RESET BUTTON PUSHED?
         $OUT[1] =TRUE     ; DOORLOCK active
         PULSE( $OUT[14], TRUE,2)      ; reset floorscanner (not yet in use)
         PULSE( $OUT[15],TRUE,2)       ; reset safety matt
         $OUT[3]=false
         $OUT[19] =TRUE 
         


      ENDIF
    ENDIF
      
    if $IN[3] THEN
      Aanvraag = true
      bWerk = false
      
      
    ENDIF
    
    IF Aanvraag and NOT runningAprogram then
      $OUT[1]=false
      $OUT[3] =False
      Aanvraag =false
    endiF

   ENDIF
    
    
;DEUR LAMPJES
IF $OUT[1] THEN
   $OUT[8] =   TRUE
   $OUT[7] =   FALSE
   $OUT[2] =   FALSE           ; RESET LAMP UIT
ELSE
   $OUT[8] =   FALSE
   $OUT[7] =   TRUE
   $OUT[2] =   true  

ENDIF
if Aanvraag  then
   $OUT[3] =TRUE
   
else
   $OUT[3] =false
ENDIF



	;ENDFOLD (BASISTECH PLC)
    ;FOLD BACKUPMANAGER PLC
    IF BM_ENABLED THEN
      BM_OUTPUTSIGNAL = BM_OUTPUTVALUE
      ENDIF
    ;ENDFOLD (BACKUPMANAGER PLC)
    ;FOLD USER PLC
    ;Make your modifications here

    ;ENDFOLD (USER PLC)
    
  ENDLOOP
  
END
  ;ENDFOLD

DEF  RESET_OUT ( )
  INT N
  MsgLoop(" ")
  IF REFLECT_PROG_NR == 1 THEN
    FOR N = 0 TO PGNO_LENGTH - 1
      $OUT[PGNO_FBIT_REFL + N] = FALSE
    ENDFOR
  ENDIF
  IF (PGNO_REQ>0) THEN
    $OUT[PGNO_REQ]=FALSE
  ELSE
    IF (PGNO_REQ<0) THEN
      $OUT[-PGNO_REQ]=TRUE
    ENDIF
  ENDIF
END

DEF RESET_LINESEL()
  $LINE_SEL_OK=FALSE
END

DEF FLIKKERLAMP(LampOut: IN, Count: IN)
  DECL int Count      
  decl int LampOut
   
   if Count > 200 then
      $OUT[LampOut]= true
   else
      $OUT[LampOut]= true

   endif
END


DEF SK()
;FOLD +> Status Key detection
; check if status key is pressed
  $FLAG[11]=is_key_pressed(14)
  $FLAG[12]=is_key_pressed(15)
  $FLAG[13]=is_key_pressed(16)
  $FLAG[14]=is_key_pressed(17)
 
; detect rising edge of the button press
; by comparing current and remembered state
  $FLAG[1] = $FLAG[11] AND NOT $FLAG[15]
  $FLAG[2] = $FLAG[12] AND NOT $FLAG[16]
  $FLAG[3] = $FLAG[13] AND NOT $FLAG[17]
  $FLAG[4] = $FLAG[14] AND NOT $FLAG[18]
;ENDFOLD
;FOLD +> Status Key control
; allow status keys to control some outputs
; but only if in T1 and drives are enables
IF $T1 AND $PERI_RDY AND $USER_SAF AND NOT $PRO_ACT THEN
   IF $FLAG[1] THEN
    $OUT[18] = NOT $OUT[18] ; clamp open
    $OUT[17] = NOT $OUT[18] ; clamp close
  ENDIF
ENDIF
 
;ENDFOLD
END
